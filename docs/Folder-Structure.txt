# This is the suggested folder structure for the BIMS project. It may change as the project progresses but the core
# structure will remain the same or at least most of it will.

/bims-project
│
├── .github/                # GitHub Actions for Auto-Builds/Watchtower
├── .envs/                  # Environment variables (.env.ce, .env.pro)
├── apps/                   # Django Apps
│   ├── core/               # Shared logic, base models (Residents, Households)
│   ├── certificates/       # PDF Generation logic (CE vs Pro templates)
│   ├── business/           # Pro Tier: Business Permits & Revenue
│   ├── blotter/            # Pro Tier: Justice & Incident logs
│   └── audit/              # Ultra Tier: Audit Logging & Security
│
├── core/                   # Django Project Settings (settings.py, wsgi.py)
├── static/                 # Tailwind CSS, Images, Alpine.js scripts
├── media/                  # Uploaded files (e.g., Resident photos)
├── templates/              # HTML Templates (organized by tier)
│   ├── base.html
│   ├── partials/           # Reusable components (e.g., sidebars, headers, footers, etc.)
│   ├── components/         # Reusable components (e.g., tables, buttons, modals, etc.)
│   └── pages/              # Page templates (organized by tier)
│
├── requirements/           # Split dependencies
│   ├── base.txt            # Django, Postgres, WhiteNoise
│   └── pro.txt             # WeasyPrint (PDF), Cryptography
│
├── docker-compose.yml
├── Dockerfile
├── manage.py
└── README.md
